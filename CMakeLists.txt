/* Polyfill getsubopt */
static __inline int getsubopt(char **optionp, char * const *tokens, char **valuep) {
    char *subopt;
    int i;

    *valuep = NULL;
    if (optionp == NULL || *optionp == NULL)
        return -1;

    subopt = *optionp;
    while (**optionp != '\0' && **optionp != ',' && **optionp != '=')
        (*optionp)++;

    if (**optionp == '=') {
        **optionp = '\0';
        (*optionp)++;
        *valuep = *optionp;
        while (**optionp != '\0' && **optionp != ',')
            (*optionp)++;
        if (**optionp == ',') {
            **optionp = '\0';
            (*optionp)++;
        }
    } else if (**optionp == ',') {
        **optionp = '\0';
        (*optionp)++;
    }

    for (i = 0; tokens[i] != NULL; i++) {
        if (strcmp(subopt, tokens[i]) == 0)
            return i;
    }

    if (*valuep == NULL)
        *valuep = subopt;

    return -1;
}

/* Polyfill setgroups */
static __inline int setgroups(size_t size, const gid_t *list) {

    return 0;
}

/* Polyfill getline */
static __inline ssize_t getline(char **lineptr, size_t *n, FILE *stream) {
    size_t pos = 0;
    int c;

    if (lineptr == NULL || n == NULL || stream == NULL) {
        errno = EINVAL;
        return -1;
    }

    if (*lineptr == NULL || *n == 0) {
        *n = 128;
        *lineptr = (char *)malloc(*n);
        if (*lineptr == NULL) {
            errno = ENOMEM;
            return -1;
        }
    }

    while ((c = fgetc(stream)) != EOF) {
        if (pos + 2 >= *n) {
            size_t new_size = *n * 2;
            char *new_ptr = (char *)realloc(*lineptr, new_size);
            if (new_ptr == NULL) {
                errno = ENOMEM;
                return -1;
            }
            *lineptr = new_ptr;
            *n = new_size;
        }
        (*lineptr)[pos++] = (char)c;
        if (c == '\n')
            break;
    }

    if (pos == 0 && c == EOF)
        return -1;

    (*lineptr)[pos] = '\0';
    return (ssize_t)pos;
}
